CREATE TABLE kyc_activity (
	id VARCHAR2 ( 64 ) NOT NULL,
	name VARCHAR2 ( 255 ) DEFAULT NULL,
	rule_desc CLOB DEFAULT NULL,
	type_code VARCHAR2 ( 64 ) DEFAULT NULL,
	type_name VARCHAR2 ( 255 ) DEFAULT NULL,
	scope_code VARCHAR2 ( 64 ) DEFAULT NULL,
	scope_name VARCHAR2 ( 255 ) DEFAULT NULL,
	issue_rule_code VARCHAR2 ( 64 ) DEFAULT NULL,
	issue_rule_name VARCHAR2 ( 255 ) DEFAULT NULL,
	start_time DATE DEFAULT NULL,
	end_time DATE DEFAULT NULL,
	status VARCHAR2 ( 64 ) DEFAULT NULL,
	create_time DATE NOT NULL,
	creator_id VARCHAR2 ( 64 ) DEFAULT NULL,
	creator_name VARCHAR2 ( 255 ) DEFAULT NULL,
	update_time DATE DEFAULT NULL,
	updater_id VARCHAR2 ( 64 ) DEFAULT NULL,
	updater_name VARCHAR2 ( 255 ) DEFAULT NULL,
	CONSTRAINT pk_kyc_activity PRIMARY KEY ( id )
);
COMMENT ON TABLE kyc_activity IS '活动基础信息表';
COMMENT ON COLUMN kyc_activity.id IS '活动ID';
COMMENT ON COLUMN kyc_activity.name IS '活动名称';
COMMENT ON COLUMN kyc_activity.rule_desc IS '活动规则描述';
COMMENT ON COLUMN kyc_activity.type_code IS '活动类型编码：01-新用户拉运首单送油气消费礼，02-装货完成送油气消费礼，03-卸货完成送油气消费礼';
COMMENT ON COLUMN kyc_activity.type_name IS '活动类型名称';
COMMENT ON COLUMN kyc_activity.scope_code IS '活动范围编码：01-指定货主，02-指定区域，03-指定货源，04-指定业务单位，05-指定业务部门，06-不限制';
COMMENT ON COLUMN kyc_activity.scope_name IS '活动范围名称';
COMMENT ON COLUMN kyc_activity.issue_rule_code IS '发放规则编码：01-每一单，02-车辆每天第一单，03-车主每天第一单，04-车辆每周第一单，05-车主每周第一单，06-车辆每月第一单，07-车主每月第一单';
COMMENT ON COLUMN kyc_activity.issue_rule_name IS '发放规则名称';
COMMENT ON COLUMN kyc_activity.start_time IS '活动开始时间';
COMMENT ON COLUMN kyc_activity.end_time IS '活动结束时间';

COMMENT ON COLUMN kyc_activity.status IS '状态：01-启用，02-禁用，03-已结束';
COMMENT ON COLUMN kyc_activity.create_time IS '创建时间';
COMMENT ON COLUMN kyc_activity.creator_id IS '创建人ID';
COMMENT ON COLUMN kyc_activity.creator_name IS '创建人姓名';
COMMENT ON COLUMN kyc_activity.update_time IS '更新时间';
COMMENT ON COLUMN kyc_activity.updater_id IS '更新人ID';
COMMENT ON COLUMN kyc_activity.updater_name IS '更新人姓名';

CREATE INDEX idx_kyc_activity_type ON kyc_activity ( type_code );
CREATE INDEX idx_kyc_activity_scope ON kyc_activity ( scope_code );
CREATE INDEX idx_kyc_activity_dist_rule ON kyc_activity ( issue_rule_code );
CREATE INDEX idx_kyc_activity_time ON kyc_activity ( start_time, end_time );
CREATE INDEX idx_kyc_activity_status ON kyc_activity ( status );

CREATE TABLE kyc_activity_gift (
	id VARCHAR2 ( 64 ) NOT NULL,
