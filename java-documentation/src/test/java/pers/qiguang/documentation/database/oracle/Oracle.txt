
在 Oracle 数据库中，可以通过 CREATE INDEX 语句为表中的某个字段创建索引。

CREATE INDEX index_name
ON table_name (column_name);

在 Oracle 数据库中，可以使用 ALTER TABLE 语句来向现有表添加新字段，并使用 COMMENT 语句为字段添加注释。

ALTER TABLE 表名
ADD (字段名 数据类型 [约束条件]);

COMMENT ON COLUMN 表名.字段名 IS '注释内容';

示例

ALTER TABLE KYC_VEHICLE_TRANSFER
ADD (APPLICANT VARCHAR2(100));

COMMENT ON COLUMN KYC_VEHICLE_TRANSFER.APPLICANT IS '申请人';


ALTER TABLE KYC_VEHICLE
ADD (VEHICLE_IMAGE_URL VARCHAR2(255));

COMMENT ON COLUMN KYC_VEHICLE.VEHICLE_IMAGE_URL IS '车辆，车身照URL';




ALTER TABLE KYC_VEHICLE
ADD (operation_type VARCHAR2(16));

COMMENT ON COLUMN KYC_VEHICLE.operation_type IS '操作类型(manual:01人工操作/auto:02自动操作/system:03系统操作';

ALTER TABLE KYC_VEHICLE
ADD (data_category VARCHAR2(16));

COMMENT ON COLUMN KYC_VEHICLE.data_category IS '数据类别(current:01当前数据/historical:02历史数据)';

ALTER TABLE KYC_VEHICLE
ADD (data_source VARCHAR2(16));

COMMENT ON COLUMN KYC_VEHICLE.data_source IS '数据来源(01 WEB/02 IOS/03 ANDROID/04 H5/05 Mini/Applet)';




CREATE TABLE sys_operation_log (
  -- 基础信息
  id VARCHAR2(64) NOT NULL,
  tenant_id VARCHAR2(64) DEFAULT NULL,
  trace_id VARCHAR2(64) DEFAULT NULL,
  log_type VARCHAR2(64) DEFAULT NULL,

  -- 操作内容
  module VARCHAR2(50) DEFAULT NULL,
  sub_module VARCHAR2(50) DEFAULT NULL,
  business_type VARCHAR2(64) DEFAULT NULL,
  operation_type VARCHAR2(20) DEFAULT NULL,
  operation_desc CLOB DEFAULT NULL,
  business_id VARCHAR2(64) DEFAULT NULL,
  business_data CLOB DEFAULT NULL,

  -- 请求信息
  request_method VARCHAR2(10) DEFAULT NULL,
  request_url VARCHAR2(500) DEFAULT NULL,
  request_params CLOB DEFAULT NULL,

  ip_location VARCHAR2(100) DEFAULT NULL,
  network_type VARCHAR2(20) DEFAULT NULL,
  user_agent CLOB DEFAULT NULL,

  -- 操作人信息
  operator_id VARCHAR2(64) DEFAULT NULL,
  operator_name VARCHAR2(50) DEFAULT NULL,
  operator_account VARCHAR2(50) DEFAULT NULL,
  operator_role VARCHAR2(100) DEFAULT NULL,
  operator_dept VARCHAR2(100) DEFAULT NULL,
  operator_org VARCHAR2(100) DEFAULT NULL,

  -- 操作状态
  status VARCHAR2(64) DEFAULT NULL,
  error_code VARCHAR2(50) DEFAULT NULL,
  error_msg CLOB DEFAULT NULL,
  error_stack CLOB DEFAULT NULL,
  retry_count VARCHAR2(64) DEFAULT NULL,

  -- 主键和约束
