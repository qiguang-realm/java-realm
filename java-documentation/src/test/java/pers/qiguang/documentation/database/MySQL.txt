SELECT
	m.NICK_NAME AS '用户昵称',
	v.VEHICLE_NUMBER AS '车牌号',
	m.MOBILE AS '手机号'
FROM
	st_user_vehicle uv
	INNER JOIN st_member m ON m.ID = uv.USER_ID
	INNER JOIN st_vehicle v ON v.VEHICLE_ID = uv.VEHICLE_ID
WHERE
	m.`STATUS` = 1
	AND v.`STATUS` = '01'
	AND uv.`STATUS` = '01'
	AND v.audit_status = '02'
	AND v.VEHICLE_NUMBER LIKE '%5%2%6%'
	AND v.VEHICLE_NUMBER REGEXP '5.*2.*6'
ORDER BY
	v.create_time DESC
