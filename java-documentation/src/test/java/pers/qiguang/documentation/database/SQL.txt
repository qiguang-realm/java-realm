
要查询数据库表中某字段是否有重复数据，并统计重复数据的数量，可以使用 GROUP BY 和 HAVING 子句来实现。以下是一个示例 SQL 查询：

SELECT column_name, COUNT(*) AS duplicate_count
FROM table_name
GROUP BY column_name
HAVING COUNT(*) > 1;

解释：
column_name: 你要检查的字段名。

table_name: 你要查询的表名。

COUNT(*): 统计每个值的出现次数。

GROUP BY column_name: 按字段值分组。

HAVING COUNT(*) > 1: 过滤出出现次数大于 1 的记录，即重复数据。

查询MySQL数据库中表的大小和占用空间

方法二：查询特定数据库中的表大小

SELECT
    table_name AS '表名',
    ROUND(data_length/1024/1024, 2) AS '数据大小(MB)',
    ROUND(index_length/1024/1024, 2) AS '索引大小(MB)',
    ROUND((data_length + index_length)/1024/1024, 2) AS '总大小(MB)',
    table_rows AS '行数'
FROM
    information_schema.TABLES
WHERE
    table_schema = '你的数据库名'
ORDER BY
    (data_length + index_length) DESC;

方法三：使用SHOW TABLE STATUS命令

SHOW TABLE STATUS FROM 你的数据库名;

DROP TABLE IF EXISTS `infra_config`;

CREATE TABLE `infra_config` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '参数主键',
  `category` varchar(50) NOT NULL COMMENT '参数分组', // 无DEFAULT，必须显式赋值
  `type` tinyint NOT NULL COMMENT '参数类型',        // 无DEFAULT，必须显式赋值
  `name` varchar(100) NOT NULL DEFAULT '' COMMENT '参数名称', // 保留DEFAULT
  `config_key` varchar(100) NOT NULL COMMENT '参数键名',     // 【改动】去掉了DEFAULT ''，必须显式赋值
  `value` varchar(2000) NOT NULL DEFAULT '' COMMENT '参数键值', // 保留DEFAULT
  `visible` bit(1) NOT NULL DEFAULT b'1' COMMENT '是否可见', // 必须保留DEFAULT
  `remark` varchar(500) DEFAULT NULL COMMENT '备注', // 允许NULL的字段，本身就不需要DEFAULT
  `creator` varchar(64) NOT NULL DEFAULT '' COMMENT '创建者', // 保留DEFAULT更安全
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updater` varchar(64) NOT NULL DEFAULT '' COMMENT '更新者', // 保留DEFAULT更安全
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `is_deleted` bit(1) NOT NULL DEFAULT b'0' COMMENT '是否删除', // 必须保留DEFAULT
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_config_key` (`config_key`, `is_deleted`),
  KEY `idx_category` (`category`),
  KEY `idx_update_time` (`update_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='参数配置表';
