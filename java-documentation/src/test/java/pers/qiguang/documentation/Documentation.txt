    DTO主要用于服务层与Controller层之间的数据传输。当后端方法需要接收复杂参数时，通过一个DTO对象来封装多个请求参数，能够提高代码的可读性和维护性。例如，在一个电子商务网站中，当用户提交订单时，Controller层会接收一个包含用户输入信息的UserDTO对象，并将其传递给Service层进行处理。

VO主要用于展示层与后端服务之间的数据交互。它封装了前端页面所需的所有数据，以满足视图层的展示需求。例如，前端页面需要显示用户信息和订单信息，这些数据会被封装在一个UserVO对象中，并传输到前端进行展示。

总的来说，在选择使用VO还是DTO时，应考虑具体的业务需求和系统架构。如果主要是处理前端传入的数据并直接用于视图展示，VO更为合适；若涉及复杂的业务逻辑和跨层数据传输，则宜使用DTO。

数据库金额字段存储最佳实践（特别是以分为单位），对应的Java实体类类型选择

总结
场景	数据库类型 (Oracle)	Java 类型	说明
存储单位（分）	NUMBER(10)	Long	避免小数，直接存整数分
业务逻辑	-	BigDecimal	需转换时使用，如分转元展示


import java.math.BigDecimal;

public class Product {
    // 数据库映射字段（以分为单位）
    private Long priceInCents;

    // 业务方法：获取元为单位的金额
    public BigDecimal getPriceInYuan() {
        return priceInCents != null
            ? new BigDecimal(priceInCents).divide(new BigDecimal(100))
            : null;
    }

    // 业务方法：设置元为单位的金额
    public void setPriceInYuan(BigDecimal priceInYuan) {
        this.priceInCents = priceInYuan != null
            ? priceInYuan.multiply(new BigDecimal(100)).longValue()
            : null;
    }

    // Getter/Setter
    public Long getPriceInCents() {
        return priceInCents;
    }

    public void setPriceInCents(Long priceInCents) {
        this.priceInCents = priceInCents;
    }
}

import java.math.BigDecimal;

public class Product {
    private Long priceInCents;  // 数据库存储的分为单位值

@ApiModelProperty(value = "券面值(单位:分，整数类型)", example = "1000")
@TableField("FACE_VALUE")
private Long faceValue;

/**
 * 获取面值对应的元单位金额（用于精确计算）
 * @return 元单位的金额字符串（保留2位小数，如"10.00"），null表示未设置
 * @apiNote 适用于需要精确金额计算的场景，返回值不带单位
 */
public String getFaceValueYuan() {
    return faceValue == null ? null :
            new BigDecimal(faceValue)
                    .divide(new BigDecimal(100), 2, RoundingMode.DOWN)
                    .toString();
}

/**
 * 设置面值（通过元单位字符串）
 * @param yuan 元单位的金额字符串（如"10.00"），null表示清空面值
 * @throws NumberFormatException 当输入不是有效数字格式时抛出
 * @apiNote 输入值将自动转换为分单位存储
 */
public void setFaceValueYuan(String yuan) {
    this.faceValue = yuan == null ? null :
            new BigDecimal(yuan).multiply(new BigDecimal(100)).longValue();
}

/**
 * 获取带单位的格式化面值展示文本（用于界面显示）
 * @return 格式化后的展示文本，包含单位和适当精度：
 *         - 折扣券(02): "8.50折"（强制2位小数）
 *         - 免息按天(04): "30天"（整数）
 *         - 免息按期(05): "3期"（整数）
 *         - 其他类型: "50.00元"（强制2位小数）
 *         null表示未设置面值
 */
public String getDisplayFaceValue() {
    if (faceValue == null) {
        return null;
    }

    BigDecimal yuanValue = new BigDecimal(faceValue)
            .divide(new BigDecimal(100), 2, RoundingMode.DOWN);

    switch (typeCode) {
        case "02":
            return String.format("%.2f折", yuanValue);
        case "04":
            return yuanValue.setScale(0, RoundingMode.DOWN) + "天";
        case "05":
            return yuanValue.setScale(0, RoundingMode.DOWN) + "期";
        default:
            return String.format("%.2f元", yuanValue);
    }
}

    // 直接访问分值的getter/setter
    public Long getPriceInCents() {
        return priceInCents;
    }

    public void setPriceInCents(Long cents) {
        this.priceInCents = cents;
    }
}
