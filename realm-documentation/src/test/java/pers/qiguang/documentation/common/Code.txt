/**
 * 检查文件名是否已存在（优化版，使用LIMIT 1提高查询效率）
 *
 * @param fileName 要检查的文件名
 * @return 如果文件名已存在返回true，否则返回false
 */
public boolean isFileNameDuplicate(String fileName) {
    if (StringUtils.isBlank(fileName)) {
        return false;
    }

    LambdaQueryWrapper<VehiclePassMileage> queryWrapper = new LambdaQueryWrapper<>();
    queryWrapper.eq(VehiclePassMileage::getOriginalFileName, fileName)
            .last("LIMIT 1");  // 关键优化点

    return baseMapper.selectOne(queryWrapper) != null;
}

public boolean isFileNameDuplicate(String fileName) {
    if (StringUtils.isBlank(fileName)) {
        return false;
    }

    LambdaQueryWrapper<VehiclePassMileage> queryWrapper = new LambdaQueryWrapper<>();
    queryWrapper.eq(VehiclePassMileage::getOriginalFileName, fileName);

    // 对于Oracle，使用selectCount并限制只查询1条记录
    return baseMapper.selectCount(queryWrapper) > 0;
}

public void writeVehicle(ListPageVehicleReqDTO reqDTO, HttpServletResponse response) {
    try {
        setExcelRespProp(response);

        // 移除分页限制，一次性查询所有数据
        reqDTO.setPageNum(1);
        reqDTO.setPageSize(Integer.MAX_VALUE); // 或者使用一个足够大的数字

        // 或者更好的方式 - 分批查询写入
        List<Vehicle> allRecords = new ArrayList<>();
        int currentPage = 1;
        int pageSize = 5000; // 适当的分批大小
        boolean hasMore = true;

        while (hasMore) {
            reqDTO.setPageNum(currentPage);
            reqDTO.setPageSize(pageSize);
            Page<Vehicle> page = vehicleService.listPageVehicle(reqDTO).getPage();
            allRecords.addAll(page.getRecords());

            hasMore = page.getRecords().size() >= pageSize;
            currentPage++;
        }

        EasyExcel.write(response.getOutputStream(), Vehicle.class)
                .sheet("车辆信息")
                .doWrite(allRecords);

        logger.info("车辆信息Excel导出成功，记录数：{}", allRecords.size());
    } catch (IOException e) {
        logger.error("导出车辆信息Excel失败，异常信息：{}", e.getMessage(), e);
    }
}


@ApiOperation(value = "导出企业车辆信息Excel文件")
@RequestMapping(value = "/exportEnterpriseVehicles", method = RequestMethod.GET)
public void exportEnterpriseVehicles(ListPageEnterpriseVehiclesReqDTO reqDTO, HttpServletResponse response) {
    // 1. 设置响应头（必须在获取 OutputStream 之前）
    try {
        response.setContentType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
        response.setCharacterEncoding("utf-8");
        String fileName = URLEncoder.encode("车辆信息", "UTF-8").replaceAll("\\+", "%20");
        response.setHeader("Content-Disposition", "attachment;filename=" + fileName + ".xlsx");
        response.setHeader("Access-Control-Expose-Headers", "Content-Disposition");
        log.info("Excel响应属性设置成功，文件名：{}", fileName + ".xlsx");
    } catch (Exception e) {
        log.error("设置响应头失败", e);
        return;
    }

    // 2. 获取数据（避免在 try-with-resources 中处理业务逻辑）
    reqDTO.setPageNum(1);
    reqDTO.setPageSize(10000);
    List<EnterpriseVehiclesDO> records = enterpriseVehiclesService.listPageEnterpriseVehicles(reqDTO).getPage().getRecords();

    // 3. 使用 try-with-resources 确保流关闭，但手动控制写入过程
    try (ServletOutputStream out = response.getOutputStream()) {
        // 关键点：使用 ExcelWriter 显式控制写入流程
        ExcelWriter excelWriter = EasyExcel.write(out)
                .head(EnterpriseVehiclesDO.class)
                .autoCloseStream(false)  // 必须设为 false，防止 EasyExcel 自动关闭流
                .build();

        try {
            WriteSheet writeSheet = EasyExcel.writerSheet("车辆信息").build();
            excelWriter.write(records, writeSheet);
            excelWriter.finish();  // 显式调用 finish() 确保数据刷出
            out.flush();           // 强制刷新缓冲区
            log.info("车辆信息Excel导出成功，记录数：{}", records.size());
        } catch (Exception e) {
            log.error("Excel写入失败", e);
            throw e; // 重新抛出异常以确保流被正确关闭
        }
    } catch (IOException e) {
        log.error("导出车辆信息Excel失败", e);
        // 注意：此处不要抛异常，否则可能触发容器级别的重复异常处理
    }
}
